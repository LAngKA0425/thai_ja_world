│   │
│   └── (main)/                 - Protected application routes
│       ├── layout.tsx          - Main app layout with top/bottom nav
│       ├── plaza/page.tsx      - Social plaza (foundation)
│       ├── shop/page.tsx       - Shop with categories and purchase
│       ├── home/page.tsx       - Mini home with stats
│       ├── friends/page.tsx    - Friends list and requests
│       └── menu/page.tsx       - Settings and menu options
│
└── api/                        - API route handlers
    ├── auth/

    │   ├── login/route.ts      - User login

    │
    ├── users/

    │           ├── route.ts    - Get minihome data

    ├── friends/

    │   └── [friendshipId]/route.ts - Accept/reject/remove
    │
    ├── shop/
    │   ├── items/route.ts      - List shop items
    │   └── purchase/route.ts   - Purchase items
    │
    ├── inventory/route.ts      - User inventory
    │
    ├── broadcast/route.ts      - Send/get broadcasts
    │
    ├── moderation/
    │   ├── report/route.ts     - Report users
    │   └── block/route.ts      - Block/unblock users
    │
    └── admin/
